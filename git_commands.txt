git clone <url> - клонирует удалённый репозиторий на локальный компьютер
git status - показывает текущее состояние репозитория (изменённые/неотслеживаемые файлы)
git add <файл> - добавляет файл в индекс (подготовку к коммиту)
git commit -m "сообщение" - создаёт коммит с зафиксированными изменениями
git push origin <ветка> - отправляет локальные коммиты на удалённый репозиторий
git pull - загружает изменения из удалённого репозитория и сливает их с текущей веткой
git branch - показывает список веток
git checkout <ветка> - переключается на другую ветку
git checkout -b <ветка> - создаёт новую ветку и переключается на неё
git merge <ветка> - сливает указанную ветку в текущую
git log - показывает историю коммитов
git diff - показывает различия между изменениями
